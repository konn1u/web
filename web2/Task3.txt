const pupils = [
    {name: "Alex", grade: 4},
    {name: "Pol", grade: 5},
    {name: "Bob", grade: 3},
    {name: "Adam", grade: 2},
    {name: "Dora", grade: 5},
    {name: "Vanya", grade: 3},
    {name: "Vitya", grade: 2},
    {name: "Pasha", grade: 4},
    {name: "Slava", grade: 2}
];

const a = prompt("Grade:");

const a_pupils = pupils
    .filter(p => p.grade > a)
    .map(p => p.name);

const sr = pupils.reduce((accumulator, current) => accumulator.grade + current.grade, 0) / pupils.length;
console.log(a_pupils, sr);
